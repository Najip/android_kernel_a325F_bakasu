+static bool susfs_is_valid_app_name(const char *app_name) {

+	size_t len = strlen(app_name);
+	if (len == 0 || len > 255)
+		return false;

+	if (!strcmp(app_name, ".") || !strcmp(app_name, ".."))
+		return false;

+			continue;

+					uid, token, SUSFS_UID_FILTER_MIN_UID,
+					SUSFS_UID_FILTER_MAX_UID);

+				return -EINVAL;
+			continue;
